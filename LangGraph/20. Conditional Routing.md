Conditional routing in LangGraph relies on builder.add_conditional_edges(source_node, routing_function, path_map). The routing function evaluates the current state and returns a target destination or key.
Below are four practical examples covering the most common real-world use cases.
1. Classification & Intent Routing
Routes a user's prompt to different downstream specialist nodes based on category classification.
from typing import TypedDict, Literal
from langgraph.graph import StateGraph, START, END

class SupportState(TypedDict):
    ticket_text: str
    category: str
    response: str

def classify_intent(state: SupportState) -> dict:
    text = state["ticket_text"].lower()
    if "refund" in text or "billing" in text:
        cat = "billing"
    elif "bug" in text or "error" in text:
        cat = "tech_support"
    else:
        cat = "general"
    return {"category": cat}

def handle_billing(state: SupportState) -> dict:
    return {"response": "Routing to Accounts & Billing department."}

def handle_tech(state: SupportState) -> dict:
    return {"response": "Routing to Tier 2 Technical Engineering."}

def handle_general(state: SupportState) -> dict:
    return {"response": "Handling via General Knowledge Base."}

# Routing function inspects the state:
def route_by_category(state: SupportState) -> Literal["billing", "tech_support", "general"]:
    return state["category"]

builder = StateGraph(SupportState)
builder.add_node("classifier", classify_intent)
builder.add_node("billing_node", handle_billing)
builder.add_node("tech_node", handle_tech)
builder.add_node("general_node", handle_general)

builder.add_edge(START, "classifier")

# Conditional edge with an explicit path_map:
builder.add_conditional_edges(
    "classifier",
    route_by_category,
    {
        "billing": "billing_node",
        "tech_support": "tech_node",
        "general": "general_node",
    }
)

builder.add_edge("billing_node", END)
builder.add_edge("tech_node", END)
builder.add_edge("general_node", END)

app = builder.compile()
print(app.invoke({"ticket_text": "I see an error code 500 when saving."})["response"])
# Output: Routing to Tier 2 Technical Engineering.

2. Cyclical Retry & Error Self-Correction Loop
Evaluates output quality or test failures, either cycling back for another retry attempt or exiting to END when passing or exhausting attempts.
from typing import TypedDict, Literal
from langgraph.graph import StateGraph, START, END

class CodeGenState(TypedDict):
    prompt: str
    code: str
    test_passed: bool
    retry_count: int

def generate_code(state: CodeGenState) -> dict:
    current_retries = state.get("retry_count", 0)
    # Simulate fixing the error on attempt 2
    passed = current_retries >= 2
    code_snippet = "def solve(): return 42" if passed else "def solve(): raise Error"
    
    return {
        "code": code_snippet,
        "test_passed": passed,
        "retry_count": current_retries + 1
    }

def evaluate_code(state: CodeGenState) -> dict:
    # Validation step
    return {}

# Router deciding loop vs. exit:
def should_retry(state: CodeGenState) -> Literal["retry", "finish"]:
    if state["test_passed"]:
        return "finish"
    if state["retry_count"] >= 3:
        print("[Router] Max retries reached, aborting.")
        return "finish"
    print(f"[Router] Tests failed. Retrying (attempt {state['retry_count']})...")
    return "retry"

builder = StateGraph(CodeGenState)
builder.add_node("generator", generate_code)
builder.add_node("evaluator", evaluate_code)

builder.add_edge(START, "generator")
builder.add_edge("generator", "evaluator")

builder.add_conditional_edges(
    "evaluator",
    should_retry,
    {
        "retry": "generator",  # Loops back!
        "finish": END
    }
)

app = builder.compile()
res = app.invoke({"prompt": "Write solver", "test_passed": False, "retry_count": 0})
print("Final code:", res["code"], "| Retries:", res["retry_count"])

3. ReAct Agent Routing via Tool Call Inspection
The standard agentic AI loop where the routing function inspects whether the model produced an action call or finished talking.
from typing import Annotated, Literal
from typing_extensions import TypedDict
from langchain_core.messages import BaseMessage, AIMessage
from langgraph.graph import StateGraph, START, END
from langgraph.graph.message import add_messages[span_4](start_span)[span_4](end_span)

class AgentState(TypedDict):
    messages: Annotated[list[BaseMessage], add_messages][span_5](start_span)[span_5](end_span)

def agent_node(state: AgentState) -> dict:
    # In real applications, this is an LLM call: response = model.invoke(state["messages"])
    return {}

def tool_node(state: AgentState) -> dict:
    return {}

# Routing function:
def router(state: AgentState) -> Literal["tools", "__end__"]:
    last_message = state["messages"][-1]
    
    # If the LLM requested a tool execution:
    if hasattr(last_message, "tool_calls") and len(last_message.tool_calls) > 0:
        return "tools"
    # Otherwise, complete the run
    return "__end__"

builder = StateGraph(AgentState)
builder.add_node("agent", agent_node)
builder.add_node("tools", tool_node)

builder.add_edge(START, "agent")
builder.add_conditional_edges("agent", router, {"tools": "tools", "__end__": END})
builder.add_edge("tools", "agent")  # Return to agent after running tool[span_6](start_span)[span_6](end_span)

4. Parallel Fan-Out (Returning a List of Destinations)
Conditional edges can return a list of strings instead of a single string. When a list is returned, LangGraph forks execution and runs all targeted branches concurrently in parallel.
import operator
from typing import Annotated, TypedDict
from langgraph.graph import StateGraph, START, END

class ResearchState(TypedDict):
    query: str
    needs_finance: bool
    needs_legal: bool
    reports: Annotated[list[str], operator.add]

def planner(state: ResearchState) -> dict:
    return {}

def run_finance_analysis(state: ResearchState) -> dict:
    return {"reports": ["Finance Analysis: Clean"]}

def run_legal_analysis(state: ResearchState) -> dict:
    return {"reports": ["Legal Analysis: Compliant"]}

# Returns multiple destination nodes simultaneously
def parallel_router(state: ResearchState) -> list[str]:
    targets = []
    if state.get("needs_finance"):
        targets.append("finance_worker")
    if state.get("needs_legal"):
        targets.append("legal_worker")
    return targets if targets else ["__end__"]

builder = StateGraph(ResearchState)
builder.add_node("planner", planner)
builder.add_node("finance_worker", run_finance_analysis)
builder.add_node("legal_worker", run_legal_analysis)

builder.add_edge(START, "planner")

# Conditional edge with multiple concurrent targets
builder.add_conditional_edges(
    "planner",
    parallel_router,
    {
        "finance_worker": "finance_worker",
        "legal_worker": "legal_worker",
        "__end__": END
    }
)

builder.add_edge("finance_worker", END)
builder.add_edge("legal_worker", END)

app = builder.compile()
output = app.invoke({"query": "Acquisition Audit", "needs_finance": True, "needs_legal": True, "reports": []})
print(output["reports"])
# Output: ['Finance Analysis: Clean', 'Legal Analysis: Compliant']
