```javascript
async function getHierarchyTree(partyId) {
    // PostgreSQL query
    const tree = await pg.query(`
        WITH RECURSIVE org_tree AS (
            -- Get user's direct ancestors
            SELECT 
                h.parent_id as id,
                p.name,
                p.party_type,
                'ancestor' as relation,
                h.depth
            FROM hierarchy h
            JOIN party p ON p.id = h.parent_id
            WHERE h.child_id = $1
            UNION ALL
            -- Get all descendants
            SELECT 
                h.child_id as id,
                p.name,
                p.party_type,
                'descendant' as relation,
                h.depth
            FROM hierarchy h
            JOIN party p ON p.id = h.child_id
            WHERE h.path @> ARRAY[$1] AND h.child_id != $1
        )
        SELECT * FROM org_tree ORDER BY depth
    `, [partyId]);
    
    return tree.rows;
}
```

## 📊 Category 2: Dashboard Metrics Endpoints

### 3. Get Main Dashboard Summary
```
GET /api/dashboard/summary
```
**Purpose:** Main dashboard landing – KPI cards with totals

**Query Params (optional filters):**
```
?period=current_month
&state=TX
&market_category=PPO
&status=approved
```

**Response:**
```json
{
  "period": "2026-08",
  "summary": {
    "total_applications": 845,
    "total_renewals": 234,
    "total_commissions": 45678.90,
    "total_premium": 1234567.89,
    "conversion_rate": 68.4,
    "avg_commission_per_app": 54.06
  },
  "trends": {
    "applications": { "current": 845, "previous": 723, "change": 16.8 },
    "commissions": { "current": 45678.90, "previous": 38900.00, "change": 17.4 }
  },
  "filters_applied": {
    "state": "TX",
    "market_category": null,
    "status": null
  },
  "last_updated": "2026-08-13T14:30:00Z"
}
```

**Implementation:**
```javascript
async function getDashboardSummary(partyId, filters) {
    // Step 1: Get hierarchy IDs
    const partyIds = await getDescendantIds(partyId);
    
    // Step 2: Build MongoDB pipeline
    const pipeline = [
        {
            $match: {
                "broker.party_id": { $in: partyIds },
                "application_date": { $gte: getPeriodStart(filters.period) },
                ...(filters.state && { "address.state": filters.state }),
                ...(filters.market_category && { "product.category": filters.market_category }),
                ...(filters.status && { "status": filters.status })
            }
        },
        {
            $facet: {
                totals: [
                    {
                        $group: {
                            _id: null,
                            total_applications: { $sum: 1 },
                            total_renewals: { 
                                $sum: { $cond: [{ $eq: ["$transaction_type", "renewal"] }, 1, 0] }
                            },
                            total_commissions: { $sum: "$commission_amount" },
                            total_premium: { $sum: "$premium_amount" }
                        }
                    }
                ],
                conversion: [
                    {
                        $group: {
                            _id: "$status",
                            count: { $sum: 1 }
                        }
                    }
                ]
            }
        }
    ];
    
    const results = await db.collection('transactions').aggregate(pipeline).toArray();
    
    // Calculate conversion rate
    const total = results[0].totals[0] || {};
    const statusCounts = results[0].conversion.reduce((acc, curr) => {
        acc[curr._id] = curr.count;
        return acc;
    }, {});
    
    return {
        summary: {
            ...total,
            conversion_rate: (statusCounts.approved / (statusCounts.approved + statusCounts.rejected) * 100) || 0
        }
    };
}
```

```javascript
async function getBreakdowns(partyId, filters) {
    const { dimension, period } = filters;
    
    // Map dimension to MongoDB field
    const dimensionMap = {
        state: "address.state",
        market_category: "product.category",
        status: "status",
        broker: "broker.party_id",
        product: "product.name"
    };
    
    const pipeline = [
        {
            $match: {
                "broker.party_id": { $in: await getDescendantIds(partyId) },
                "application_date": { $gte: getPeriodStart(period) }
            }
        },
        {
            $group: {
                _id: `$${dimensionMap[dimension]}`,
                count: { $sum: 1 },
                commissions: { $sum: "$commission_amount" },
                premium: { $sum: "$premium_amount" }
            }
        },
        { $sort: { count: -1 } },
        { $limit: 10 }
    ];
    
    return await db.collection('transactions').aggregate(pipeline).toArray();
}
```

**Implementation:**
```javascript
async function drillDown(partyId, filters) {
    const { dimension, key, sub_dimension, period } = filters;
    
    // Re-run aggregation with additional filter
    const matchStage = {
        "broker.party_id": { $in: await getDescendantIds(partyId) },
        "application_date": { $gte: getPeriodStart(period) },
        [dimensionMap[dimension]]: key  // Add the selected dimension
    };
    
    // Get sub-breakdown
    const subPipeline = [
        { $match: matchStage },
        {
            $group: {
                _id: `$${dimensionMap[sub_dimension]}`,
                count: { $sum: 1 },
                commissions: { $sum: "$commission_amount" }
            }
        },
        { $sort: { count: -1 } }
    ];
    
    return await db.collection('transactions').aggregate(subPipeline).toArray();
}
```

```javascript
async function getAvailableFilters(partyId) {
    const partyIds = await getDescendantIds(partyId);
    
    // Get distinct values from MongoDB
    const [states, categories, statuses] = await Promise.all([
        db.collection('transactions').distinct('address.state', { "broker.party_id": { $in: partyIds } }),
        db.collection('transactions').distinct('product.category', { "broker.party_id": { $in: partyIds } }),
        db.collection('transactions').distinct('status', { "broker.party_id": { $in: partyIds } })
    ]);
    
    // Get team members from PostgreSQL
    const brokers = await pg.query(`
        SELECT p.id, p.name 
        FROM party p
        JOIN hierarchy h ON h.child_id = p.id
        WHERE h.path @> ARRAY[$1] AND p.party_type = 'INDIVIDUAL'
    `, [partyId]);
    
    return { states, market_categories: categories, statuses, brokers: brokers.rows };
}
```

## 🚀 Complete UI Rendering Flow

### Frontend Workflow:
```javascript
class BrokerDashboard {
    async load() {
        // 1. Get user permissions
        const permissions = await api.get('/permissions');
        this.renderControls(permissions);
        
        // 2. Load organization tree for sidebar
        const tree = await api.get('/hierarchy/tree');
        this.renderOrgTree(tree);
        
        // 3. Load available filters
        const filters = await api.get('/filters');
        this.populateFilterDropdowns(filters);
        
        // 4. Load main dashboard
        await this.loadDashboard();
    }
    
    async loadDashboard(filters = {}) {
        // Load summary KPIs
        const summary = await api.get('/dashboard/summary', { params: filters });
        this.updateKPIs(summary);
        
        // Load breakdown charts
        const breakdowns = await api.get('/dashboard/breakdowns', { 
            params: { ...filters, dimension: 'state' }
        });
        this.renderChart('state_chart', breakdowns);
        
        // Load trends
        const trends = await api.get('/dashboard/trends', { 
            params: { ...filters, metric: 'applications' }
        });
        this.renderLineChart('trend_chart', trends);
    }
    
    async drillDown(dimension, key) {
        const data = await api.get('/dashboard/drilldown', {
            params: { 
                dimension, 
                key, 
                sub_dimension: 'market_category',
                ...this.currentFilters 
            }
        });
        this.showDrillDownModal(data);
    }
}
```

## 🔐 Security Considerations

1. **All endpoints must validate:**
```javascript
// Middleware to ensure user can only access their hierarchy
async function validateAccess(req, res, next) {
    const { party_id } = req.user;
    const targetIds = req.body.party_ids || [req.params.partyId];
    
    // Check if all target IDs are descendants of user
    const hasAccess = await pg.query(`
        SELECT COUNT(*) > 0 as has_access
        FROM hierarchy h
        WHERE h.path @> ARRAY[$1] 
          AND h.child_id = ANY($2)
    `, [party_id, targetIds]);
    
    if (!hasAccess.rows[0].has_access) {
        return res.status(403).json({ error: 'Access denied' });
    }
    next();
}
```

2. **Cache invalidation:**
```javascript
// Invalidate cache when data changes
async function invalidateDashboardCache(partyIds) {
    for (const id of partyIds) {
        const keys = await redis.keys(`dashboard:${id}:*`);
        if (keys.length) await redis.del(keys);
    }
}
```
