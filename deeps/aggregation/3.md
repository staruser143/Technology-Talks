## The Complete Pipeline

```javascript
// Step 1: Create a view that federates external data
// Run this once to set up virtual collections for your external sources

// If using Atlas Data Federation, create virtual collections first:
// - external.agencyHierarchy (points to S3/HTTP)
// - external.brokerTiers (points to S3/HTTP)

// Step 2: The main aggregation pipeline - run every 5 minutes
db.dashboard_metrics.aggregate([

  // Stage 1: Pull transactions and join with broker tiers
  {
    $lookup: {
      from: "external.brokerTiers",
      localField: "brokerId",
      foreignField: "brokerId",
      as: "brokerTierInfo"
    }
  },
  { $unwind: { path: "$brokerTierInfo", preserveNullAndEmptyArrays: true } },

  // Stage 2: Join with agency hierarchy
  {
    $lookup: {
      from: "external.agencyHierarchy",
      localField: "brokerId",
      foreignField: "brokerId",
      as: "agencyInfo"
    }
  },
  { $unwind: { path: "$agencyInfo", preserveNullAndEmptyArrays: true } },

  // Stage 3: Add date-based grouping keys
  {
    $addFields: {
      dateKey: { $dateToString: { format: "%Y-%m-%d", date: "$createdAt" } },
      monthKey: { $dateToString: { format: "%Y-%m", date: "$createdAt" } },
      quarterKey: {
        $concat: [
          { $toString: { $year: "$createdAt" } },
          "-Q",
          { $toString: { $ceil: { $divide: [{ $month: "$createdAt" }, 3] } } }
        ]
      }
    }
  },

  // Stage 4: Group by multiple dimensions for dashboard widgets
  {
    $facet: {
      
      // Widget 1: Daily totals by broker tier
      dailyTierMetrics: [
        { $group: {
          _id: {
            date: "$dateKey",
            tier: "$brokerTierInfo.tier",
            type: "$type"
          },
          totalAmount: { $sum: "$amount" },
          transactionCount: { $sum: 1 },
          avgAmount: { $avg: "$amount" }
        }},
        { $sort: { "_id.date": -1, "totalAmount": -1 } }
      ],

      // Widget 2: Agency hierarchy performance
      agencyPerformance: [
        { $group: {
          _id: {
            agencyId: "$agencyInfo.agencyId",
            agencyName: "$agencyInfo.agencyName",
            parentAgencyId: "$agencyInfo.parentAgencyId",
            region: "$agencyInfo.region"
          },
          totalPremium: { $sum: "$amount" },
          applicationCount: { 
            $sum: { $cond: [{ $eq: ["$type", "application"] }, 1, 0] }
          },
          renewalCount: { 
            $sum: { $cond: [{ $eq: ["$type", "renewal"] }, 1, 0] }
          }
        }},
        { $sort: { "totalPremium": -1 } }
      ],

      // Widget 3: Monthly trend by broker
      brokerMonthlyTrend: [
        { $group: {
          _id: {
            brokerId: "$brokerId",
            month: "$monthKey",
            tier: "$brokerTierInfo.tier"
          },
          monthlyTotal: { $sum: "$amount" },
          transactionCount: { $sum: 1 }
        }},
        { $sort: { "_id.month": -1, "monthlyTotal": -1 } }
      ],

      // Widget 4: High-level summary (for dashboard header)
      summaryMetrics: [
        { $group: {
          _id: null,
          totalRevenue: { $sum: "$amount" },
          totalTransactions: { $sum: 1 },
          uniqueBrokers: { $addToSet: "$brokerId" },
          applications: { 
            $sum: { $cond: [{ $eq: ["$type", "application"] }, 1, 0] }
          },
          renewals: { 
            $sum: { $cond: [{ $eq: ["$type", "renewal"] }, 1, 0] }
          }
        }},
        { $project: {
          _id: 0,
          totalRevenue: 1,
          totalTransactions: 1,
          uniqueBrokerCount: { $size: "$uniqueBrokers" },
          applications: 1,
          renewals: 1,
          renewalRate: {
            $cond: [
              { $eq: ["$totalTransactions", 0] },
              0,
              { $divide: ["$renewals", "$totalTransactions"] }
            ]
          }
        }}
      ],

      // Widget 5: Top 10 brokers
      topBrokers: [
        { $group: {
          _id: "$brokerId",
          totalAmount: { $sum: "$amount" },
          transactionCount: { $sum: 1 },
          tier: { $first: "$brokerTierInfo.tier" },
          agencyName: { $first: "$agencyInfo.agencyName" }
        }},
        { $sort: { totalAmount: -1 } },
        { $limit: 10 }
      ]
    }
  },

  // Stage 5: Flatten the facet results into a single document
  {
    $project: {
      _id: 0,
      generatedAt: new Date(),
      data: {
        dailyTierMetrics: 1,
        agencyPerformance: 1,
        brokerMonthlyTrend: 1,
        summaryMetrics: 1,
        topBrokers: 1
      }
    }
  },

  // Stage 6: Merge into your dashboard collection
  {
    $merge: {
      into: "dashboard_metrics",
      on: "generatedAt", // Will create a new document every run
      whenMatched: "replace",
      whenNotMatched: "insert"
    }
  }
]);
```

## Setting Up the Scheduled Job

**Option A: MongoDB Atlas Triggers (Serverless)**
```javascript
// Create a scheduled trigger in Atlas that runs every 5 minutes
exports = function() {
  const pipeline = [ /* the entire pipeline from above */ ];
  const database = context.services.get("mongodb-atlas").db("your_db");
  return database.collection("transactions").aggregate(pipeline).toArray();
};
```

**Option B: Node.js Cron Job**
```javascript
const cron = require('node-cron');
const { MongoClient } = require('mongodb');

cron.schedule('*/5 * * * *', async () => {
  console.log('Refreshing dashboard data...');
  const client = await MongoClient.connect(process.env.MONGO_URI);
  const db = client.db('your_db');
  
  await db.collection('transactions').aggregate([ /* pipeline */ ]).toArray();
  console.log('Dashboard data refreshed at', new Date());
  await client.close();
});
```

  const latest = await db.collection('dashboard_metrics')
    .find({})
    .sort({ generatedAt: -1 })

## Key Optimizations

1. **Index your transaction collection** on `brokerId`, `createdAt`, and `type` for faster source queries.
2. **Partition external data** by date in S3 (e.g., `s3://broker-data/2026/08/20/`) to reduce scan time.
3. **Use change streams** instead of cron if you need near-real-time updates (sub-minute).
